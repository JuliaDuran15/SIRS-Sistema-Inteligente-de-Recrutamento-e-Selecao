━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
 0. LOGIN — obter token JWT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Login como admin (substitua as credenciais conforme .env)
curl -s -X POST http://localhost:8000/auth/token \
  -d "username=admin@sirs.com&password=admin123" \
  -H "Content-Type: application/x-www-form-urlencoded" | python3 -m json.tool

# Copie o valor de "access_token" e cole abaixo:
TOKEN="COLE_O_TOKEN_AQUI"

# Atalho para testar se o token funciona:
curl -s http://localhost:8000/auth/me \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

 1. USUÁRIOS — criar e listar

# Criar usuário RH
curl -s -X POST http://localhost:8000/usuarios/ \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \

    "nome": "Joana RH Teste",
    "email": "joana.rh@teste.com",

    "nome": "Pedro Gestor Teste",
    "email": "pedro.gestor@teste.com",

  }' | python3 -m json.tool

# Listar todos os usuários
curl -s http://localhost:8000/usuarios/ \

 2. VAGAS — criar, listar, editar pesos
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Criar vaga (admin/RH)
curl -s -X POST http://localhost:8000/vagas/ \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "nome": "Desenvolvedor Python Pleno",
    "requisitos_texto": "Desenvolvedor Python com 3 anos de experiência em FastAPI, SQLAlchemy e PostgreSQL. Conhecimento de Docker e CI/CD. Inglês para leitura técnica.",
    "gestores_ids": ["'"$GESTOR_ID"'"],
    "peso_rh": 0.6,
    "peso_mercado": 0.4,
    "peso_curriculo": 0.5,
    "peso_entrevista_rh": 0.25,
    "peso_entrevista_tec": 0.25
  }' | python3 -m json.tool

# Listar vagas
curl -s http://localhost:8000/vagas/ \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

# Copie o id da vaga criada:
VAGA_ID="COLE_O_ID_DA_VAGA"

# Ver detalhes da vaga
curl -s http://localhost:8000/vagas/$VAGA_ID \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

# Editar pesos da vaga (recalcula scores de todos os CVs)
curl -s -X PATCH http://localhost:8000/vagas/$VAGA_ID/pesos \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "peso_rh": 0.7,
    "peso_mercado": 0.3,
    "peso_curriculo": 0.6,
    "peso_entrevista_rh": 0.2,
    "peso_entrevista_tec": 0.2
  }' | python3 -m json.tool

# Editar nome e requisitos da vaga (revectoriza + recalcula scores)
curl -s -X PATCH http://localhost:8000/vagas/$VAGA_ID/requisitos \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "nome": "Desenvolvedor Python Sênior",
    "requisitos_texto": "Python sênior com 5+ anos. FastAPI, SQLAlchemy, PostgreSQL, Docker, AWS, microsserviços. Liderança técnica e inglês fluente."
  }' | python3 -m json.tool

# Alterar status da vaga (aberta → fechada)
curl -s -X PATCH "http://localhost:8000/vagas/$VAGA_ID/status?status=fechada" \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

 4. CANDIDATURAS — criar, enviar currículo, triagem
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Criar candidatura (vincula candidato à vaga)
curl -s -X POST http://localhost:8000/candidaturas/ \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "candidato_id": "'"$CANDIDATO_ID"'",
    "vaga_id": "'"$VAGA_ID"'"
  }' | python3 -m json.tool

# Copie o id da candidatura:
CANDIDATURA_ID="COLE_O_ID_DA_CANDIDATURA"

# Enviar currículo como texto (sem PDF — processa via Celery)
curl -s -X POST http://localhost:8000/candidaturas/$CANDIDATURA_ID/curriculo/texto \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "texto": "Desenvolvedor Python com 6 anos de experiência. Especialista em FastAPI, SQLAlchemy, PostgreSQL, Docker e AWS. Liderou migração de monolito para microsserviços. Inglês fluente."
  }' | python3 -m json.tool

# Ver candidatura (aguarde alguns segundos para o Celery processar o CV)
curl -s http://localhost:8000/candidaturas/$CANDIDATURA_ID \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

# Listar candidaturas da vaga
curl -s "http://localhost:8000/candidaturas/?vaga_id=$VAGA_ID" \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

# Aprovar na triagem
curl -s -X PATCH "http://localhost:8000/candidaturas/$CANDIDATURA_ID/status?novo_status=aprovado_triagem&ator=Joana+RH" \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

 5. ENTREVISTAS — agendar, registrar resultado
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# Agendar entrevista RH (candidatura deve estar em aprovado_triagem)
curl -s -X POST http://localhost:8000/entrevistas/ \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "candidatura_id": "'"$CANDIDATURA_ID"'",
    "tipo": "rh",
    "agendada_para": "2026-07-10T14:00:00"
  }' | python3 -m json.tool

# Copie o id da entrevista:
ENTREVISTA_ID="COLE_O_ID_DA_ENTREVISTA"

# Registrar resultado da entrevista RH
curl -s -X PATCH http://localhost:8000/entrevistas/$ENTREVISTA_ID/resultado \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "score_manual": 8.5,
    "anotacoes": "Candidato comunicativo, demonstrou boa profundidade técnica em Python e experiência sólida com APIs REST. Apresentou cases reais de arquitetura de microsserviços.",
    "pontos_fortes": ["comunicação clara", "experiência com microsserviços", "inglês fluente"],
    "pontos_fracos": ["pouca experiência com Kubernetes"]
  }' | python3 -m json.tool

# Listar entrevistas da candidatura
curl -s http://localhost:8000/entrevistas/candidatura/$CANDIDATURA_ID \
  -H "Authorization: Bearer $TOKEN" | python3 -m json.tool

# Agendar entrevista técnica (após RH realizada)
curl -s -X POST http://localhost:8000/entrevistas/ \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "candidatura_id": "'"$CANDIDATURA_ID"'",
    "tipo": "tecnica",
    "agendada_para": "2026-07-15T10:00:00"
  }' | python3 -m json.tool
